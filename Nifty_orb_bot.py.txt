def fetch_1min_candles_raw() -> pd.DataFrame | None:
    try:
        data = dhan.intraday_minute_data(
            security_id=NIFTY_SECURITY_ID, exchange_segment="IDX_I", instrument_type="INDEX"
        )
        if not data or data.get("status") != "success":
            return None

        df = pd.DataFrame(data["data"])
        df["start_time"] = pd.to_datetime(df["start_time"])
        df = df[df["start_time"].dt.date == datetime.date.today()]
        df = df.sort_values("start_time").reset_index(drop=True)

        for col in ["open", "high", "low", "close"]:
            if col in df.columns:
                df[col] = pd.to_numeric(df[col], errors="coerce")
        return df
    except Exception as e:
        log.error(f"fetch_1min_candles_raw() error: {e}")
        return None

def resample_to_15min(df: pd.DataFrame) -> pd.DataFrame | None:
    try:
        df = df.set_index("start_time")
        df_15 = df.resample("15min", offset="15min").agg({
            "open": "first", "high": "max", "low": "min", "close": "last"
        }).dropna(subset=["open", "close"])
        return df_15.reset_index().rename(columns={"index": "start_time"})
    except Exception as e:
        log.error(f"resample_to_15min() error: {e}")
        return None

def fetch_latest_15min_close() -> tuple[float, float, float] | None:
    raw = fetch_1min_candles_raw()
    if raw is None or raw.empty: return None
    df = resample_to_15min(raw)
    if df is None or df.empty: return None

    cutoff = datetime.datetime.now() - datetime.timedelta(minutes=15)
    completed = df[df["start_time"] <= cutoff]
    if completed.empty: return None

    last = completed.iloc[-1]
    return float(last["close"]), float(last["high"]), float(last["low"])

def extract_orb_range():
    raw = fetch_1min_candles_raw()
    if raw is None or raw.empty: return

    raw_orb = raw[(raw["start_time"].dt.time >= datetime.time(9, 30)) & (raw["start_time"].dt.time < datetime.time(10, 0))]
    if len(raw_orb) < 28:
        log.warning(f"Incomplete morning range tracking array profile ({len(raw_orb)} bars). Retrying.")
        return

    df_15 = resample_to_15min(raw)
    if df_15 is None or df_15.empty: return

    orb_candles = df_15[(df_15["start_time"].dt.time >= datetime.time(9, 30)) & (df_15["start_time"].dt.time < datetime.time(10, 0))]
    if len(orb_candles) < 2: return

    state["orb_high"] = float(orb_candles["high"].max())
    state["orb_low"]  = float(orb_candles["low"].min())
    state["box_size"] = state["orb_high"] - state["orb_low"]

    log.info(f"ORB Box calculated → High: {state['orb_high']} | Low: {state['orb_low']}")
    send_telegram_alert(f"📦 *ORB Box Formed*\n🟢 High: {state['orb_high']}\n🔴 Low: {state['orb_low']}\n📏 Size: {state['box_size']:.2f} pts")

def get_nifty_expiry_from_api() -> str | None:
    try:
        response = dhan.get_expiry_list(underlying_scrip="NIFTY", exchange_segment="IDX_I")
        if not response or response.get("status") != "success": return None
        raw_dates = response.get("data", {}).get("ExpiryDate", [])
        if not raw_dates: return None

        today = datetime.date.today()
        expiry_dates = sorted(list(set([pd.to_datetime(d).date() for d in raw_dates])))
        market_closed = datetime.datetime.now().time() > datetime.time(15, 30)

        for expiry in expiry_dates:
            if expiry < today or (expiry == today and market_closed): continue
            return expiry.strftime("%Y-%m-%d")
        return None
    except Exception as e:
        log.error(f"get_nifty_expiry_from_api() fault: {e}")
        return None

def load_option_security_ids(spot_price: float):
    try:
        expiry_str = get_nifty_expiry_from_api()
        if not expiry_str: return False

        chain = dhan.get_option_chain(underlying_scrip="NIFTY", underlying_seg="IDX_I", expiry=expiry_str)
        if not chain or chain.get("status") != "success": return False

        atm_strike = round(spot_price / 50) * 50
        call_strike = atm_strike - 50   # 1 Strike ITM
        put_strike  = atm_strike + 50   

        options = chain["data"]
        for opt in options:
            if opt["strikePrice"] == call_strike and opt["optionType"] == "CALL":
                state["call_security_id"] = str(opt["securityId"])
            if opt["strikePrice"] == put_strike and opt["optionType"] == "PUT":
                state["put_security_id"] = str(opt["securityId"])

        return state["call_security_id"] is not None and state["put_security_id"] is not None
    except Exception as e:
        log.error(f"load_option_security_ids() processing exception: {e}")
        return False

def execute_market_order(transaction_type: str, quantity: int, option_type: str) -> bool:
    security_id = state["call_security_id"] if option_type == "CALL" else state["put_security_id"]
    if not security_id: return False

    send_telegram_alert(f"⏳ *Routing Order to Dhan API...*\nAction: {transaction_type}\nQty: {quantity}\nContract Type: {option_type}")

    try:
        response = dhan.place_order(
            security_id=security_id, exchange_segment=dhan.NSE_FNO,
            transaction_type=dhan.BUY if transaction_type == "BUY" else dhan.SELL,
            quantity=quantity, order_type=dhan.MARKET, product_type="INTRADAY", price=0
        )

        if response and response.get("status") == "success":
            send_telegram_alert(f"✅ *ORDER EXECUTED SUCCESS*\nResponse: {response.get('message', 'Filled')}")
            return True
        else:
            send_telegram_alert(f"❌ *ORDER REJECTED BY BROKER*\nPayload: {response}")
            return False
    except Exception as api_fault:
        send_telegram_alert(f"🚨 *CRITICAL EXECUTION EXCEPTION*\nFault Trace: {api_fault}")
        return False

    elif pos == "SHORT":

    elif pos == "REV_SHORT":
        execute_market_order("SELL", REVERSAL_LOTS * QTY_PER_LOT, "PUT")
    state["active_position"] = "FINISHED"
